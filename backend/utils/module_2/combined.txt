"""
Data loader for factory layout - loads from data.json
"""

import json
import os
from dataclasses import dataclass
from typing import List, Tuple

@dataclass
class Machine:
    """
    Machine definition for factory layout
    
    Attributes:
        name: Model name (e.g., "Wood Cutter")
        model: Exact model identifier (e.g., "WC-3000X")
        count: Number of machines of this type
        use: Description of what the machine does
        image: URL or path to single picture of the machine
        width: Width in meters
        height: Height in meters
    """
    name: str
    model: str
    count: int
    use: str
    image: str
    width: float  # meters
    height: float  # meters


def load_data(json_path: str = "data.json"):
    """Load machine data from JSON file (flows are now generated by LLM)"""
    # Get the directory of this file
    current_dir = os.path.dirname(os.path.abspath(__file__))
    full_path = os.path.join(current_dir, json_path)
    
    with open(full_path, 'r', encoding='utf-8') as f:
        data = json.load(f)
    
    # Convert machine dictionaries to Machine objects
    machines = [Machine(**m) for m in data['machines']]
    
    return machines

class Timer:
    def __init__(self):
        self.t0 = time.time()
        self.steps = []

    def log(self, name):
        now = time.time()
        duration = now - self.t0
        self.steps.append((name, duration))
        print(f"[{name}] took {duration:.2f}s")
        self.t0 = now

    def print_summary(self):
        print("\n" + "=" * 30)
        print("TIMING SUMMARY")
        print("=" * 30)
        total = 0
        for name, duration in self.steps:
            print(f"{name:<30}: {duration:.2f}s")
            total += duration
        print("-" * 30)
        print(f"{'TOTAL':<30}: {total:.2f}s")
        print("=" * 30)


def encode_image(image_path):
    with open(image_path, "rb") as image_file:
        return base64.b64encode(image_file.read()).decode("utf-8")


def ask_stop_step():
    while True:
        print("\nINTERACTIVE MODE")
        print("Stop at which step?")
        print("1) After pipeline/flows")
        print("2) After layout generation")
        print("3) After base DXF+PNG export")
        print("4) After AI realistic render")
        c = input("Choose (1-4): ").strip()
        if c in ("1", "2", "3", "4"):
            return int(c)


def _call_remove_all_machine_backgrounds(images_dir: str, workers: int = 14):
    """
    Calls image_processor.remove_all_machine_backgrounds in a signature-tolerant way.

    Requirement:
    - Go through ALL images in ./images
    - Process IN PARALLEL (workers=14)
    - Replace each image with its cleaned (background-removed) variant (overwrite)
    """
    images_dir = str(Path(images_dir).resolve())
    if not os.path.isdir(images_dir):
        print(f"[WARN] images dir not found: {images_dir} (skipping background removal)")
        return

    fn = remove_all_machine_backgrounds
    sig = inspect.signature(fn)
    params = sig.parameters

    kwargs = {}

    # Try common parameter names
    if "images_dir" in params:
        kwargs["images_dir"] = images_dir
    elif "input_dir" in params:
        kwargs["input_dir"] = images_dir
    elif "folder" in params:
        kwargs["folder"] = images_dir
    elif "dir_path" in params:
        kwargs["dir_path"] = images_dir

    if "workers" in params:
        kwargs["workers"] = workers
    elif "max_workers" in params:
        kwargs["max_workers"] = workers
    elif "num_workers" in params:
        kwargs["num_workers"] = workers

    # Force overwrite if supported
    if "overwrite" in params:
        kwargs["overwrite"] = True
    if "replace_originals" in params:
        kwargs["replace_originals"] = True
    if "inplace" in params:
        kwargs["inplace"] = True

    # If it expects no args, just call it (but still meets "first thing" requirement)
    try:
        if kwargs:
            result = fn(**kwargs)
        else:
            result = fn()

        # Best-effort post-handling for common return types:
        # - dict {src: out}
        # - list of (src, out)
        # - list of output paths
        # If outputs are separate and overwrite wasn't supported, replace originals when possible.
        if result and not (
            ("overwrite" in params) or ("replace_originals" in params) or ("inplace" in params)
        ):
            try:
                mapping = None
                if isinstance(result, dict):
                    mapping = result
                elif isinstance(result, list) and result and isinstance(result[0], (tuple, list)) and len(result[0]) == 2:
                    mapping = dict(result)

                if mapping:
                    for src, out in mapping.items():
                        if src and out and os.path.exists(out) and os.path.exists(src):
                            shutil.copyfile(out, src)
                # If it's just a list of cleaned paths, we can't reliably map -> skip.
            except Exception:
                pass

        print(f"[+] Background removal done for images in: {images_dir} (workers={workers})")
    except TypeError as e:
        # If signature mismatch, try simplest (positional) call
        try:
            fn(images_dir, workers)
            print(f"[+] Background removal done for images in: {images_dir} (workers={workers})")
        except Exception as e2:
            print(f"[WARN] remove_all_machine_backgrounds failed: {e2}")
    except Exception as e:
        print(f"[WARN] remove_all_machine_backgrounds failed: {e}")


def main():
    # ✅ VERY FIRST THING: clean ALL ./images in parallel (workers=14)
    script_dir = Path(__file__).resolve().parent
    images_dir = script_dir / "images"
    _call_remove_all_machine_backgrounds(str(images_dir), workers=14)

    ap = argparse.ArgumentParser()
    ap.add_argument("--data", default="data.json")
    ap.add_argument("--links", default=os.path.join(OUTPUT_DIR, "links.json"))
    ap.add_argument("--seed", type=int, default=7)
    ap.add_argument("--i", "--interactive", action="store_true")
    args = ap.parse_args()

    stop_at = ask_stop_step() if args.i else None

    os.makedirs(OUTPUT_DIR, exist_ok=True)
    os.makedirs(PUBLIC_MODELS_DIR, exist_ok=True)

    timer = Timer()

    print("\n[*] Loading data...")
    machines = load_data(args.data)
    expanded = expand_machines(machines)
    print(f"[+] {len(expanded)} machine instances loaded")

    print("\n[*] Pipeline...")
    if should_regenerate_links(args.links, True):
        m_list_str = format_machine_list(machines)
        resp = generate_message(
            prompt=MACHINE_PIPELINE_PROMPT.format(machines_list=m_list_str),
            system_instruction="You are a factory systems engineer."
        )
        body = resp.split("```json")[-1].split("```")[0]
        pipeline = json.loads(body)
        with open("links.json", "w") as f:
            json.dump(pipeline, f, indent=2)
    else:
        pipeline = load_pipeline(args.links)

    model_flows = normalize_flows(pipeline["flows"])
    flows = expand_flows_to_instances(model_flows, machines)

    if stop_at == 1:
        print("[STOP] step 1")
        return

    print("\n[*] Placement...")
    n = compute_grid_size(expanded)
    grid, placed = place_all(expanded, flows, n, seed=args.seed)
    best, _, best_paths = local_improve(placed, flows, n, seed=args.seed + 1)
    grid_final = rebuild_grid(n, best)
    _, paths_final = ripup_reroute(grid_final, best, flows)

    if stop_at == 2:
        print("[STOP] step 2")
        return

    print("\n[*] Export base layout...")
    export_layout_with_existing_utils(best, paths_final, n, CELL_MM, flows, OUTPUT_PNG, OUTPUT_DXF)

    if stop_at == 3:
        print("[STOP] step 3")
        return

    print("\n[*] AI realistic render...")
    layout_b64 = encode_image(OUTPUT_PNG)
    realistic = generate_image(prompt=TOP_VIEW_REALISTIC_BELTS, images=[layout_b64])[0]
    realistic_path = os.path.join(OUTPUT_DIR, "realistic_layout.png")
    with open(realistic_path, "wb") as f:
        f.write(realistic)

    if stop_at == 4:
        print("[STOP] step 4")
        return

    print("\n[*] Remove background...")
    clean = remove_background(realistic)
    clean_path = os.path.join(OUTPUT_DIR, "realistic_clean.png")
    with open(clean_path, "wb") as f:
        f.write(clean)

    print("\n[*] Generate 3D...")
    with open(clean_path, "rb") as f:
        result = generate_3d(images=[f])
    url = result["model_file"]
    r = requests.get(url)
    with open(FINAL_GLB_PATH, "wb") as f:
        f.write(r.content)

    print(f"[SUCCESS] Saved: {FINAL_GLB_PATH}")
    timer.print_summary()


if __name__ == "__main__":
    main()

def ensure_layers(doc):
    """Create all necessary layers for the DXF document"""
    layers = [
        ("WALL_OUT", DXF_BLACK),
        ("WALL_IN",  DXF_BLACK),
        ("WALL_HATCH", DXF_BLUE),
        ("PILLARS", DXF_BLACK),
        ("DOORS", DXF_BLACK),
        ("GRID", DXF_BLACK),
        ("DIMS", DXF_BLACK),
        ("TEXT", DXF_BLACK),
        ("LEGEND", DXF_BLACK),
        ("MACHINES", DXF_BLUE),
        ("DETAILS", DXF_BLUE),
        ("FLOW", DXF_BLUE),
        ("IMAGES", DXF_BLUE),
        ("GROUPS", 6),
    ]
    for lname, color in layers:
        if lname not in doc.layers:
            doc.layers.new(lname, dxfattribs={"color": color})


def ensure_linetypes(doc):
    """Create custom linetypes for the DXF document"""
    if "DASHED" not in doc.linetypes:
        doc.linetypes.new("DASHED", dxfattribs={"description":"Dashed", "pattern":[0.8, 0.4, -0.4]})
    if "CENTER" not in doc.linetypes:
        doc.linetypes.new("CENTER", dxfattribs={"description":"Center", "pattern":[1.0, 0.6, -0.2, 0.2, -0.2]})


def draw_lwpoly(msp, pts, layer, closed=False, width=0, linetype=None):
    """Draw a lightweight polyline"""
    dxf = {"layer": layer, "closed": closed}
    if width and width > 0:
        dxf["const_width"] = width
    if linetype:
        dxf["linetype"] = linetype
    msp.add_lwpolyline(pts, dxfattribs=dxf)


def draw_rect(msp, x, y, w, h, layer, linetype=None):
    """Draw a rectangle"""
    pts = [(x,y),(x+w,y),(x+w,y+h),(x,y+h),(x,y)]
    draw_lwpoly(msp, pts, layer, closed=True, width=0, linetype=linetype)


def draw_pillar(msp, cx, cy, size=PILLAR_SIZE_MM):
    """Draw a pillar at the specified center point"""
    half = size/2
    draw_rect(msp, cx-half, cy-half, size, size, layer="PILLARS")


def draw_wall_hatch(msp, outer: Tuple[float,float,float,float], inner: Tuple[float,float,float,float], step=WALL_HATCH_STEP):
    """
    Simple hatch approximation between outer and inner rectangles using diagonal short segments (blue).
    This keeps it professional-ish while staying DXF-simple.
    """
    ox1, oy1, ox2, oy2 = outer
    ix1, iy1, ix2, iy2 = inner

    # draw diagonal strokes along the band (top, bottom, left, right)
    # bottom band
    yb1, yb2 = oy1, iy1
    x = ox1
    while x < ox2:
        msp.add_line((x, yb1), (x+step*0.6, yb2), dxfattribs={"layer":"WALL_HATCH"})
        x += step

    # top band
    yt1, yt2 = iy2, oy2
    x = ox1
    while x < ox2:
        msp.add_line((x, yt1), (x+step*0.6, yt2), dxfattribs={"layer":"WALL_HATCH"})
        x += step

    # left band
    xl1, xl2 = ox1, ix1
    y = oy1
    while y < oy2:
        msp.add_line((xl1, y), (xl2, y+step*0.6), dxfattribs={"layer":"WALL_HATCH"})
        y += step

    # right band
    xr1, xr2 = ix2, ox2
    y = oy1
    while y < oy2:
        msp.add_line((xr1, y), (xr2, y+step*0.6), dxfattribs={"layer":"WALL_HATCH"})
        y += step


def draw_outer_walls_arch(msp, x1, y1, x2, y2):
    """
    Proper architectural walls:
    - outer rectangle line
    - inner offset rectangle line (wall thickness)
    - entrance gap on BOTH outer and inner bottom edges
    - door leaves + arcs opening outward
    - 4 pillars on corners
    """
    t = WALL_THICK_MM

    # Pillars at corners
    draw_pillar(msp, x1, y1)
    draw_pillar(msp, x2, y1)
    draw_pillar(msp, x2, y2)
    draw_pillar(msp, x1, y2)

    # Door gap centered on bottom
    cx = (x1 + x2) / 2.0
    dL = cx - DOOR_WIDTH_MM/2
    dR = cx + DOOR_WIDTH_MM/2

    # Outer wall: draw as lines (not thick strokes)
    # bottom split
    msp.add_line((x1, y1), (dL, y1), dxfattribs={"layer":"WALL_OUT"})
    msp.add_line((dR, y1), (x2, y1), dxfattribs={"layer":"WALL_OUT"})
    # right/top/left full
    msp.add_line((x2, y1), (x2, y2), dxfattribs={"layer":"WALL_OUT"})
    msp.add_line((x2, y2), (x1, y2), dxfattribs={"layer":"WALL_OUT"})
    msp.add_line((x1, y2), (x1, y1), dxfattribs={"layer":"WALL_OUT"})

    # Inner wall rectangle offset
    ix1, iy1, ix2, iy2 = x1+t, y1+t, x2-t, y2-t
    # inner door gap (shifted up by thickness)
    idL = dL
    idR = dR
    inner_y = iy1

    # inner bottom split
    msp.add_line((ix1, inner_y), (idL, inner_y), dxfattribs={"layer":"WALL_IN"})
    msp.add_line((idR, inner_y), (ix2, inner_y), dxfattribs={"layer":"WALL_IN"})
    # inner right/top/left
    msp.add_line((ix2, iy1), (ix2, iy2), dxfattribs={"layer":"WALL_IN"})
    msp.add_line((ix2, iy2), (ix1, iy2), dxfattribs={"layer":"WALL_IN"})
    msp.add_line((ix1, iy2), (ix1, iy1), dxfattribs={"layer":"WALL_IN"})

    # Hatch between outer and inner
    draw_wall_hatch(msp, (x1,y1,x2,y2), (ix1,iy1,ix2,iy2), step=WALL_HATCH_STEP)

    # Door: double leaf opening outward (below outer wall)
    hingeL = (dL, y1)
    hingeR = (dR, y1)
    leaf = DOOR_WIDTH_MM/2

    # leaf lines outward (down)
    msp.add_line(hingeL, (dL + leaf*0.85, y1 - DOOR_DEPTH_MM), dxfattribs={"layer":"DOORS"})
    msp.add_line(hingeR, (dR - leaf*0.85, y1 - DOOR_DEPTH_MM), dxfattribs={"layer":"DOORS"})

    # swing arcs outward
    msp.add_arc(center=hingeL, radius=leaf*0.9, start_angle=270, end_angle=360, dxfattribs={"layer":"DOORS"})
    msp.add_arc(center=hingeR, radius=leaf*0.9, start_angle=180, end_angle=270, dxfattribs={"layer":"DOORS"})


def draw_room_partition_walls(msp, r: Room):
    """
    Interior partitions: use double-lines (outer+inner) lightly, no hatch.
    Keeps it clean & "real".
    """
    t = int(WALL_THICK_MM * 0.65)
    x1, y1, x2, y2 = r.x, r.y, r.x+r.w, r.y+r.h
    ix1, iy1, ix2, iy2 = x1+t, y1+t, x2-t, y2-t

    draw_rect(msp, x1, y1, x2-x1, y2-y1, layer="GROUPS")
    draw_rect(msp, ix1, iy1, ix2-ix1, iy2-iy1, layer="GROUPS", linetype="DASHED")


def draw_machine_symbol_vector(msp, doc, p: PlacedMachine):
    """Draw machine symbol with vector graphics"""
    # Footprint outline (blue)
    draw_rect(msp, p.x, p.y, p.w, p.h, layer="MACHINES")

    # Details (blue): control panel + barrels (gives top-view feel)
    cp_w = max(320, int(p.w*0.12))
    cp_h = max(220, int(p.h*0.18))
    draw_rect(msp, p.x + p.w - cp_w - 140, p.y + 140, cp_w, cp_h, layer="DETAILS")

    r = 130
    msp.add_circle((p.x + 260, p.y + 280), r, dxfattribs={"layer":"DETAILS"})
    msp.add_circle((p.x + 560, p.y + 280), r, dxfattribs={"layer":"DETAILS"})

    # Optional photo embedding layer (OFF by default)
    if EMBED_IMAGES_IN_CAD and p.img_path and os.path.exists(p.img_path):
        try:
            im = Image.open(p.img_path)
            im = ImageOps.exif_transpose(im)
            px_w, px_h = im.size
            s = min(p.w/px_w, p.h/px_h)

def generate_dxf(rooms, flows, find_machine_func, routing_func, out_dxf):
    """
    Generate complete factory DXF with walls, machines, and conveyors.
    
    Args:
        rooms: List[Room] with placed machines
        flows: List of (src_uid, dst_uid) tuples
        find_machine_func: Function to find machine by UID
        routing_func: Function to route between two machines (returns list of (x,y) points)
        out_dxf: Output DXF file path
    """
    # Create new DXF document
    doc = ezdxf.new(setup=True)
    doc.units = units.MM
    msp = doc.modelspace()
    
    # Setup layers and linetypes
    ensure_layers(doc)
    ensure_linetypes(doc)
    
    # Calculate canvas size from rooms
    if not rooms:
        canvas_w, canvas_h = 40000, 30000
    else:
        # Find bounds of all machines
        all_machines = []
        for r in rooms:
            all_machines.extend(r.machines)
        
        if all_machines:
            min_x = min(m.x for m in all_machines)
            min_y = min(m.y for m in all_machines)
            max_x = max(m.x + m.w for m in all_machines)
            max_y = max(m.y + m.h for m in all_machines)
            
            canvas_w = max(40000, int(max_x - min_x + 10000))
            canvas_h = max(30000, int(max_y - min_y + 10000))
        else:
            canvas_w, canvas_h = 40000, 30000
    
    # Draw ground (walls, grid, dimensions)
    draw_ground(doc, msp, canvas_w, canvas_h)
    
    # Draw all machines
    for room in rooms:
        # Draw room partitions if multiple rooms
        if len(rooms) > 1:
            draw_room_partition_walls(msp, room)
        
        # Draw each machine
        for machine in room.machines:
            draw_machine_symbol_vector(msp, doc, machine)
            draw_small_labels(msp, machine)
    
    # Draw flows (conveyors)
    for src_uid, dst_uid in flows:
        src = find_machine_func(rooms, src_uid)
        dst = find_machine_func(rooms, dst_uid)
        
        if src and dst:
            # Get routing waypoints
            pts = routing_func(src, dst)
            if pts and len(pts) >= 2:
                draw_double_flow(msp, pts)
    
    # Save DXF
    os.makedirs(os.path.dirname(out_dxf) or ".", exist_ok=True)
    doc.saveas(out_dxf)
    print(f"[+] DXF saved: {out_dxf}")

def export_dxf_square(
    placed: List[Placed],
    paths: Dict[Tuple[str,str], Path],
    n: int,
    out_path: str,
) -> None:
    """Export layout as DXF file with real dimensions."""
    doc = ezdxf.new(setup=True)
    doc.units = units.MM
    msp = doc.modelspace()

    # layers
    if "MACHINES" not in doc.layers:
        doc.layers.add("MACHINES")
    if "BELTS" not in doc.layers:
        doc.layers.add("BELTS")
    if "BOUNDARY" not in doc.layers:
        doc.layers.add("BOUNDARY")

    # boundary square in mm
    size_mm = to_mm(n)
    msp.add_lwpolyline([(0,0),(size_mm,0),(size_mm,size_mm),(0,size_mm),(0,0)], dxfattribs={"layer":"BOUNDARY"})

    # machines (rects with real dimensions)
    for p in placed:
        x1 = to_mm(p.x)
        y1 = to_mm(p.y)
        x2 = to_mm(p.x + p.w)
        y2 = to_mm(p.y + p.h)
        msp.add_lwpolyline([(x1,y1),(x2,y1),(x2,y2),(x1,y2),(x1,y1)], dxfattribs={"layer":"MACHINES"})
        # Add centered text label
        cx = x1 + (x2-x1)/2
        cy = y1 + (y2-y1)/2
        msp.add_text(p.uid, dxfattribs={
            "height": 500, 
            "layer": "MACHINES"
        }).set_placement(
            (cx, cy),
            align=ezdxf.enums.TextEntityAlignment.MIDDLE_CENTER
        )

    # belts
    for _, path in paths.items():
        pts = [(to_mm(x), to_mm(y)) for (x,y) in path]
        if len(pts) >= 2:
            msp.add_lwpolyline(pts, dxfattribs={"layer":"BELTS"})

    os.makedirs(os.path.dirname(out_path) or ".", exist_ok=True)
    doc.saveas(out_path)
    print(f"[+] DXF saved: {out_path}")

"""
Grid occupancy management for placement.
"""

from typing import Optional, Tuple, List
from models import Placed
from config import KEEP_OUT


class OccupancyGrid:
    """
    Cell occupancy for machines + keepout.
    Belts are routed on EDGES (separate structure).
    """
    def __init__(self, n: int):
        self.n = n
        # 0: free, 1: blocked (machine or keepout)
        self.block = [[0] * n for _ in range(n)]
        self.machine_id = [[None] * n for _ in range(n)]

    def inside(self, x: int, y: int) -> bool:
        return 0 <= x < self.n and 0 <= y < self.n

    def rect_free(self, x: int, y: int, w: int, h: int) -> bool:
        if x < 0 or y < 0 or x + w > self.n or y + h > self.n:
            return False
        for yy in range(y, y + h):
            row = self.block[yy]
            for xx in range(x, x + w):
                if row[xx] != 0:
                    return False
        return True

    def _mark(self, x: int, y: int, w: int, h: int, val: int, mid: Optional[str]):
        for yy in range(y, y + h):
            for xx in range(x, x + w):
                if 0 <= xx < self.n and 0 <= yy < self.n:
                    self.block[yy][xx] = val
                    self.machine_id[yy][xx] = mid

    def place_machine(self, p: Placed) -> bool:
        """Place machine area + keepout ring as blocked."""
        x, y, w, h = p.x, p.y, p.w, p.h
        k = KEEP_OUT
        # machine area must be free; keepout can also be free (we block it too)
        if not self.rect_free(x - k, y - k, w + 2 * k, h + 2 * k):
            return False
        # mark keepout
        self._mark(x - k, y - k, w + 2 * k, h + 2 * k, 1, None)
        # mark machine cells with id (kept blocked too)
        self._mark(x, y, w, h, 1, p.uid)
        return True

    def remove_machine(self, p: Placed, placed_all: List[Placed]) -> None:
        """
        Slow but safe: rebuild block grid from scratch (used in local search).
        For speed, you can implement incremental undo later.
        """
        self.block = [[0] * self.n for _ in range(self.n)]
        self.machine_id = [[None] * self.n for _ in range(self.n)]
        for q in placed_all:
            if q.uid == p.uid:
                continue
            self.place_machine(q)

    def hole_metric(self) -> int:
        """
        Simple hole penalty: count enclosed free components not touching boundary.
        """
        n = self.n
        seen = [[False]*n for _ in range(n)]
        dirs = [(1,0),(-1,0),(0,1),(0,-1)]

        def bfs(sx, sy) -> Tuple[int, bool]:
            stack = [(sx, sy)]
            seen[sy][sx] = True
            size = 0
            touches_border = False
            while stack:
                x, y = stack.pop()
                size += 1
                if x == 0 or y == 0 or x == n-1 or y == n-1:
                    touches_border = True
                for dx, dy in dirs:
                    nx, ny = x+dx, y+dy
                    if 0 <= nx < n and 0 <= ny < n and not seen[ny][nx] and self.block[ny][nx] == 0:
                        seen[ny][nx] = True
                        stack.append((nx, ny))
            return size, touches_border

        holes = 0
        for y in range(n):
            for x in range(n):
                if self.block[y][x] == 0 and not seen[y][x]:
                    size, touches = bfs(x, y)
                    if not touches:
                        holes += size
        return holes

"""
Utility to remove backgrounds from all machine images in the cache.

import os

def remove_all_machine_backgrounds(machines) -> Dict[str, str]:
    """
    Process all machine images to remove backgrounds.
    
    Args:
        machines: List of Machine objects with image URLs
        
    Returns:
        Dictionary mapping original image path to nobg image path
    """
    processed = {}
    
    for m in machines:
        if not m.image:
            continue
            
        # Get cached image path
        cached_path = cache_path_for(m.image)
        if not os.path.exists(cached_path):
            continue
        
        # Check if nobg version already exists
        nobg_path = cached_path.replace(".png", "_nobg.png")
        
        if os.path.exists(nobg_path):
            print(f"[*] Background already removed: {m.model}")
            processed[cached_path] = nobg_path
            continue
        
        # Remove background
        try:
            print(f"[*] Removing background: {m.model}")
            with open(cached_path, "rb") as f:
                img_bytes = f.read()
            
            nobg_bytes = remove_background(img_bytes)
            
            with open(nobg_path, "wb") as f:
                f.write(nobg_bytes)
            
            processed[cached_path] = nobg_path
            print(f"[+] Saved: {nobg_path}")
            
        except Exception as e:
            print(f"[!] Failed to remove background for {m.model}: {e}")
            processed[cached_path] = cached_path  # Use original
    
    return processed

"""

def _hash(s: str) -> str:
    return hashlib.sha256(s.encode("utf-8")).hexdigest()[:16]


def cache_path_for(filename_or_url: str) -> str:
    os.makedirs(ASSETS_DIR, exist_ok=True)
    # If it's a URL, hash it. If it's a filename, keep the name but ensure png ext
    if filename_or_url.startswith("http"):
        return os.path.join(ASSETS_DIR, f"img_{_hash(filename_or_url)}.png")
    else:
        # It is a local filename, copy it to cache for uniformity
        name = os.path.basename(filename_or_url)
        return os.path.join(ASSETS_DIR, name)


def download_image_to_png(source: str) -> Optional[str]:
    """
    Resolves an image source to a local PNG path.
    1. Checks backend/images/
    2. Checks assets_cache/
    3. Downloads if URL
    """
    if not source:
        return None

    # 1. Check Local Images Folder (backend/images)
    local_path = os.path.join(LOCAL_IMAGES_DIR, source)
    if os.path.exists(local_path):
        return local_path
    
    # 2. Check if it is a full path provided
    if os.path.exists(source):
        return source

    # 3. Setup Cache Path
    target_path = cache_path_for(source)
    if os.path.exists(target_path):
        return target_path

    # 4. Download if URL
    if source.startswith("http"):
        headers = {
            "User-Agent": "Mozilla/5.0",
            "Accept": "image/*,*/*;q=0.8",
        }
        try:
            r = requests.get(source, headers=headers, timeout=15)
            r.raise_for_status()
            with open(target_path, "wb") as f:
                f.write(r.content)
            
            # Ensure PNG
            try:
                img = Image.open(target_path).convert("RGBA")
                img.save(target_path, format="PNG")
            except Exception:
                pass
                
            return target_path
        except Exception as e:
            print(f"[!] Image download failed: {source} => {e}")
            return None
    
    print(f"[!] Image not found locally or remotely: {source}")
    return None


def load_data(path: str) -> List[Machine]:
    """Load machines from data.json"""
    with open(path, "r", encoding="utf-8") as f:

    if isinstance(raw, dict) and "machines" in raw:
        items = raw["machines"]
    elif isinstance(raw, list):
        items = raw
    else:
        raise ValueError("data.json must be either a list of machines or an object with 'machines' key")
    
    out: List[Machine] = []
    for item in items:
        out.append(
            Machine(
                model=str(item.get("model") or item.get("name") or "Machine"),
                name=str(item.get("name") or item.get("model") or ""),
                width=float(item.get("width", 1.0)),
                height=float(item.get("height", 1.0)),

                # Expecting filename in 'image' field now, e.g., "oven.png"
                image=str(item.get("image", "")), 
                use=str(item.get("use", "")),
            )
        )
    return out

        dst_count = machine_counts.get(dst_model, 1)

        if dst_count == 1: dst_instances = [dst_model]
        else: dst_instances = [f"{dst_model}_{i+1}" for i in range(dst_count)]
        
        pairs = min(len(src_instances), len(dst_instances))
        for i in range(pairs):
            expanded.append((src_instances[i], dst_instances[i]))
        
        if len(src_instances) > len(dst_instances):
            for i in range(pairs, len(src_instances)):
                expanded.append((src_instances[i], dst_instances[0]))
        elif len(dst_instances) > len(src_instances):
            for i in range(pairs, len(dst_instances)):
                expanded.append((src_instances[0], dst_instances[i]))
    
    return expanded

"""
Bridge module connecting grid layout system to existing drawing utilities.

This module converts grid-based layout data (Placed objects) into the format
expected by plot_utils and dwg_utils for visualization.

import os

def grid_to_placed_machines(placed: List[Placed], cell_mm: float = CELL_MM) -> List[PlacedMachine]:
    """
    Convert grid Placed objects to dwg_utils PlacedMachine objects.
    
    Args:
        placed: List of Placed objects from grid layout
        cell_mm: Millimeters per grid cell
        
    Returns:
        List of PlacedMachine objects with mm coordinates
    """
    machines = []
    for p in placed:
        # Prefer background-removed image if available (base_nobg.png)
        chosen_img = p.img_path
        try:
            if p.img_path:
                base, ext = os.path.splitext(p.img_path)
                nobg_candidate = f"{base}_nobg.png"
                if os.path.exists(nobg_candidate):
                    chosen_img = nobg_candidate

        except Exception:
            chosen_img = p.img_path

        pm = PlacedMachine(
            m=p.m,
            x=int(p.x * cell_mm),
            y=int(p.y * cell_mm),
            w=int(p.w * cell_mm),
            h=int(p.h * cell_mm),
            img_path=chosen_img
        )
        # Store UID for matching in find_machine
        pm.uid = p.uid
        machines.append(pm)
    return machines


def grid_to_rooms(placed: List[Placed], n: int, cell_mm: float = CELL_MM) -> List[Room]:
    """
    Convert grid layout to Room structure.
    
    Creates a single "Factory" room containing all machines.
    
    Args:
        placed: List of Placed objects from grid layout
        n: Grid size (n x n cells)
        cell_mm: Millimeters per grid cell
        
    Returns:
        List containing single Room with all machines
    """
    machines = grid_to_placed_machines(placed, cell_mm)
    
    # Calculate room bounds from grid size
    room_size_mm = int(n * cell_mm)
    
    room = Room(
        name="Factory",
        x=0,
        y=0,
        w=room_size_mm,
        h=room_size_mm,
        machines=machines
    )
    
    return [room]


def find_machine(rooms: List[Room], uid: str) -> Optional[PlacedMachine]:
    """
    Find a machine by UID across all rooms.
    
    Args:
        rooms: List of Room objects
        uid: Machine unique identifier
        
    Returns:
        PlacedMachine if found, None otherwise
    """
    for room in rooms:
        for machine in room.machines:
            # Check the uid attribute we set during conversion
            if hasattr(machine, 'uid') and machine.uid == uid:
                return machine
            # Fallback to model name for single-instance machines
            if machine.m.model == uid:
                return machine
    return None


def route_machines(a: PlacedMachine, b: PlacedMachine) -> List[Tuple[float, float]]:
    """
    Create a simple routing path between two machines.
    
    For now, uses direct routing from center to center.
    Can be enhanced later with A* pathfinding.
    
    Args:
        a: Source machine
        b: Destination machine
        
    Returns:
        List of (x_mm, y_mm) waypoints
    """
    # Calculate centers
    ax = a.x + a.w / 2.0
    ay = a.y + a.h / 2.0
    bx = b.x + b.w / 2.0
    by = b.y + b.h / 2.0
    
    # Simple L-shaped routing (horizontal then vertical)
    # This avoids diagonal lines which look less industrial
    return [
        (ax, ay),
        (bx, ay),  # Horizontal segment
        (bx, by)   # Vertical segment
    ]

"""
Local optimization with simulated annealing.
"""

import math
import random
from typing import List, Tuple, Dict

from models import Placed
from grid import OccupancyGrid
from routing import EdgeReservation, ripup_reroute, Path
from config import (
    LOCAL_ITERS, LOCAL_MAX_SHIFT, ANNEAL_START_T, ANNEAL_END_T,
    BBOX_AREA_W, HOLE_W, FLOW_LEN_W, CONGESTION_W
)


def total_congestion(res: EdgeReservation) -> int:
    """Calculate total edge congestion."""
    return sum(max(0, u - 1) for u in res.use.values())


def total_flow_len(paths: Dict[Tuple[str,str], Path]) -> int:
    """Calculate total flow path length."""
    return sum(len(p) for p in paths.values())


def layout_score(grid: OccupancyGrid, placed: List[Placed], res: EdgeReservation, paths: Dict[Tuple[str,str], Path]) -> float:
    """Calculate overall layout quality score."""
    from placement import bbox_area
    
    bb = bbox_area(placed)
    holes = grid.hole_metric()
    fl = total_flow_len(paths)
    cong = total_congestion(res)
    return (
        BBOX_AREA_W * bb
        + HOLE_W * holes * 40.0
        + FLOW_LEN_W * fl * 120.0
        + CONGESTION_W * cong * 400.0
    )


def rebuild_grid(n: int, placed: List[Placed]) -> OccupancyGrid:
    """Rebuild occupancy grid from scratch."""
    g = OccupancyGrid(n)
    for p in placed:
        if not g.place_machine(p):
            raise RuntimeError("Rebuild failed (overlap).")
    return g


def local_improve(
    placed: List[Placed],
    flows: List[Tuple[str,str]],
    n: int,
    seed: int = 9
) -> Tuple[List[Placed], EdgeReservation, Dict[Tuple[str,str], Path]]:
    """Local improvement with simulated annealing."""
    rng = random.Random(seed)

    # Make a working copy to mutate
    placed = [Placed(p.m, p.uid, p.x, p.y, p.w, p.h, p.rot, p.img_path) for p in placed]

    grid = rebuild_grid(n, placed)
    res, paths = ripup_reroute(grid, placed, flows)
    best_score = layout_score(grid, placed, res, paths)
    # Deep copy for best
    best = [Placed(p.m, p.uid, p.x, p.y, p.w, p.h, p.rot, p.img_path) for p in placed]
    best_res, best_paths = res, paths

    def temperature(t: int) -> float:
        if LOCAL_ITERS <= 1:
            return ANNEAL_END_T
        alpha = t / (LOCAL_ITERS - 1)
        return ANNEAL_START_T * (1 - alpha) + ANNEAL_END_T * alpha

    for it in range(LOCAL_ITERS):
        T = temperature(it)

        # propose move
        idx = rng.randrange(len(placed))
        p = placed[idx]
        old = Placed(p.m, p.uid, p.x, p.y, p.w, p.h, p.rot, p.img_path)

        # random move or rotate
        if rng.random() < 0.15 and p.w != p.h:
            # rotate
            p.w, p.h = p.h, p.w
            p.rot ^= 1
        else:
            dx = rng.randint(-LOCAL_MAX_SHIFT, LOCAL_MAX_SHIFT)
            dy = rng.randint(-LOCAL_MAX_SHIFT, LOCAL_MAX_SHIFT)
            p.x = max(0, min(n - p.w, p.x + dx))
            p.y = max(0, min(n - p.h, p.y + dy))

        # Quick validation: check if new position would cause overlap
        # Build a temporary grid without this machine to test
        temp_grid = OccupancyGrid(n)
        valid_move = True
        for q in placed:
            if q.uid == p.uid:
                continue
            if not temp_grid.place_machine(q):
                # This shouldn't happen with current placement, but check anyway
                valid_move = False
                break
        
        # Now try to place the moved machine
        if valid_move:
            valid_move = temp_grid.place_machine(p)
        
        if not valid_move:
            # Revert and skip this iteration
            placed[idx] = old
            continue

        # rebuild occupancy with validated placement
        try:
            grid2 = rebuild_grid(n, placed)
            res2, paths2 = ripup_reroute(grid2, placed, flows)
        except RuntimeError:
            # Rebuild failed (shouldn't happen now, but safety check)
            placed[idx] = old
            continue

        # require most flows routed (hard constraint)
        target_count = len([f for f in flows if f[0] in {q.uid for q in placed} and f[1] in {q.uid for q in placed}])
        routed = len(paths2)
        if routed < max(1, int(0.95 * target_count)):
            placed[idx] = old
            continue

        s2 = layout_score(grid2, placed, res2, paths2)
        delta = s2 - best_score

        accept = False
        if s2 <= best_score:
            accept = True
        else:
            # anneal
            prob = math.exp(-delta / max(1e-9, T * 5000.0))
            if rng.random() < prob:
                accept = True

        if accept:
            grid, res, paths = grid2, res2, paths2
            if s2 < best_score:
                best_score = s2
                best = [Placed(q.m, q.uid, q.x, q.y, q.w, q.h, q.rot, q.img_path) for q in placed]
                best_res, best_paths = res2, paths2
        else:
            placed[idx] = old

    return best, best_res, best_paths

def ensure_dir(p: str):
    os.makedirs(p, exist_ok=True)


def to_b64(img_bytes: bytes) -> str:
    return base64.b64encode(img_bytes).decode('utf-8')


def composite_on_green(png_bytes: bytes):
    buf = io.BytesIO(png_bytes)
    im = Image.open(buf).convert('RGBA')
    bg = Image.new('RGBA', im.size, (0, 255, 0, 255))
    bg.paste(im, (0, 0), im)
    out_buf = io.BytesIO()
    bg.save(out_buf, format='PNG')
    return out_buf.getvalue()

"""
Machine placement logic with greedy heuristic.
"""

import os
import math
import random
from typing import List, Tuple, Dict, Set, Optional

from models import Machine, Placed
from grid import OccupancyGrid
from io_utils import download_image_to_png
from config import (
    CELL_MM, INPUT_DIMENSIONS_IN_METERS, MIN_FOOTPRINT, MAX_FOOTPRINT,
    KEEP_OUT, CANDIDATES_PER_MACHINE, BBOX_AREA_W, FLOW_LEN_W
)


def dim_to_mm(v: float) -> float:
    """Convert dimension to mm based on input unit setting."""
    return v * 1000.0 if INPUT_DIMENSIONS_IN_METERS else v


def quantize_footprint(m: Machine) -> Tuple[int, int]:
    """Convert machine dimensions to grid cells."""
    w_mm = dim_to_mm(m.width)
    h_mm = dim_to_mm(m.height)
    cw = max(MIN_FOOTPRINT, min(MAX_FOOTPRINT, int(round(w_mm / CELL_MM))))
    ch = max(MIN_FOOTPRINT, min(MAX_FOOTPRINT, int(round(h_mm / CELL_MM))))
    return cw, ch


def expand_machines(machines: List[Machine]) -> List[Tuple[Machine, str]]:
    """Expand machines with count > 1 into individual instances."""
    out: List[Tuple[Machine, str]] = []
    for m in machines:
        if m.count <= 1:
            out.append((m, m.model))
        else:
            for i in range(m.count):
                out.append((m, f"{m.model}_{i+1}"))
    return out


def build_graph(flows: List[Tuple[str, str]]) -> Dict[str, Set[str]]:
    """Build adjacency graph from flows."""
    g: Dict[str, Set[str]] = {}
    for a, b in flows:
        g.setdefault(a, set()).add(b)
        g.setdefault(b, set()).add(a)
    return g


def bbox_of(placed: List[Placed]) -> Tuple[int,int,int,int]:
    """Calculate bounding box of placed machines."""
    if not placed:
        return (0,0,0,0)
    minx = min(p.x for p in placed)
    miny = min(p.y for p in placed)
    maxx = max(p.x + p.w for p in placed)
    maxy = max(p.y + p.h for p in placed)
    return (minx, miny, maxx, maxy)


def bbox_area(placed: List[Placed]) -> int:
    """Calculate bounding box area."""
    minx, miny, maxx, maxy = bbox_of(placed)
    return max(0, (maxx - minx) * (maxy - miny))


def estimate_flow_cost(placed_by_id: Dict[str, Placed], uid: str, cand: Placed, flows: List[Tuple[str,str]]) -> float:
    """Estimate flow cost for a candidate placement."""
    cost = 0.0
    cx, cy = cand.center()
    for a, b in flows:
        if a == uid and b in placed_by_id:
            px, py = placed_by_id[b].center()
            cost += abs(cx - px) + abs(cy - py)
        elif b == uid and a in placed_by_id:
            px, py = placed_by_id[a].center()
            cost += abs(cx - px) + abs(cy - py)
    return cost


def candidate_positions(grid: OccupancyGrid, w: int, h: int, k: int, rng: random.Random, budget: int) -> List[Tuple[int,int]]:
    """Generate candidate positions for machine placement."""
    n = grid.n
    cands: List[Tuple[int,int]] = []

    # 1) biased random samples
    tries = max(30, budget // 2)
    for _ in range(tries):
        x = rng.randint(0, max(0, n - w))
        y = rng.randint(0, max(0, n - h))
        if grid.rect_free(x - k, y - k, w + 2*k, h + 2*k):
            cands.append((x,y))
            if len(cands) >= budget:
                return cands

    # 2) scan a coarse grid
    step = max(1, int(math.sqrt(n)))
    for y in range(0, n - h + 1, max(1, step // 2)):
        for x in range(0, n - w + 1, max(1, step // 2)):
            if grid.rect_free(x - k, y - k, w + 2*k, h + 2*k):
                cands.append((x,y))
                if len(cands) >= budget:
                    return cands

    return cands


def place_all(
    expanded: List[Tuple[Machine,str]],
    flows: List[Tuple[str,str]],
    n: int,
    seed: int = 7,
) -> Tuple[OccupancyGrid, List[Placed]]:
    """Greedy placement of all machines."""
    rng = random.Random(seed)
    grid = OccupancyGrid(n)

    g = build_graph(flows)

    # order machines: area desc then degree desc
    tmp = []
    for m, uid in expanded:
        w, h = quantize_footprint(m)
        deg = len(g.get(uid, set()))
        tmp.append((w*h, deg, m, uid))
    tmp.sort(key=lambda t: (t[0], t[1]), reverse=True)

    placed: List[Placed] = []
    placed_by_id: Dict[str, Placed] = {}

    for _, _, m, uid in tmp:
        base_w, base_h = quantize_footprint(m)
        # allow rotation (swap)
        variants = [(base_w, base_h, 0)]
        if base_w != base_h:
            variants.append((base_h, base_w, 1))

        best: Optional[Placed] = None
        best_score = 1e18

        for vw, vh, rot in variants:
            cands = candidate_positions(grid, vw, vh, KEEP_OUT, rng, CANDIDATES_PER_MACHINE)
            for (x, y) in cands:
                # Prefer nobg version if it exists
                img_path = download_image_to_png(m.image)
                if img_path:
                    nobg_path = img_path.replace(".png", "_nobg.png")
                    if os.path.exists(nobg_path):
                        img_path = nobg_path
                
                cand = Placed(m=m, uid=uid, x=x, y=y, w=vw, h=vh, rot=rot, img_path=img_path)
                # score
                flow_est = estimate_flow_cost(placed_by_id, uid, cand, flows)
                test_list = placed + [cand]
                bb = bbox_area(test_list)
                score = FLOW_LEN_W * flow_est + BBOX_AREA_W * bb * 0.05
                if score < best_score:
                    if grid.rect_free(x - KEEP_OUT, y - KEEP_OUT, vw + 2*KEEP_OUT, vh + 2*KEEP_OUT):
                        best_score = score
                        best = cand

        if best is None:
            raise RuntimeError(f"Failed to place machine {uid}. Increase N or reduce footprints/keepout.")

        ok = grid.place_machine(best)
        if not ok:
            raise RuntimeError(f"Internal placement error for {uid}.")

        placed.append(best)
        placed_by_id[uid] = best

    return grid, placed


def compute_grid_size(expanded: List[Tuple[Machine,str]]) -> int:
    """Compute required grid size based on machine footprints."""
    total = 0
    for m, _ in expanded:
        w, h = quantize_footprint(m)
        total += (w + 2*KEEP_OUT) * (h + 2*KEEP_OUT)
    # fill factor provides room for belts and less holes
    fill_factor = 1.9
    n = int(math.ceil(math.sqrt(total * fill_factor)))
    # pad
    n = max(n, 12)
    return n

def rect_bounds(x, y, w, h):
    """Calculate rectangle bounds"""
    return (x, y, x + w, y + h)

def compute_bounds_rects(rects):
    """Compute overall bounds from a list of rectangles"""
    if not rects:
        return 0, 0, 1, 1
    minx = min(r[0] for r in rects)
    miny = min(r[1] for r in rects)
    maxx = max(r[2] for r in rects)
    maxy = max(r[3] for r in rects)
    return minx, miny, maxx, maxy

def trim_whitespace(im):
    """
    Automatically crops white or transparent borders from an image.
    This ensures the machine occupies the maximum visual space inside its bounding box.
    """
    try:
        # If the image has an alpha channel, use it to find the bounding box
        if im.mode in ('RGBA', 'LA'):
            bg = Image.new(im.mode, im.size, (255, 255, 255, 0))
            diff = ImageChops.difference(im, bg)
            bbox = diff.getbbox()
            if bbox:
                return im.crop(bbox)
        
        # Fallback for RGB/L (or if alpha didn't help): looks for white border
        bg = Image.new(im.mode, im.size, im.getpixel((0,0)))
        diff = ImageChops.difference(im, bg)
        diff = ImageChops.add(diff, diff, 2.0, -100)
        bbox = diff.getbbox()
        if bbox:
            return im.crop(bbox)
            
        return im
    except Exception:
        # If trimming fails, return original
        return im

def _draw_conveyor_belt(ax, pts: List[Tuple[float, float]], width_mm: float):
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]

    # Main thick belt
    ax.plot(xs, ys, color=BELT_COLOR_MAIN, linewidth=width_mm / 40, solid_capstyle="round", zorder=5)
    # Edge detail
    ax.plot(xs, ys, color=BELT_COLOR_EDGE, linewidth=max(2, width_mm / 160), solid_capstyle="round", zorder=6)

    # Rollers
    step = 3000.0
    for i in range(len(pts) - 1):
        x1, y1 = pts[i]
        x2, y2 = pts[i + 1]
        seg_len = math.hypot(x2 - x1, y2 - y1)
        if seg_len < 1:
            continue
        n = max(1, int(seg_len // step))
        for k in range(1, n + 1):
            t = (k * step) / seg_len
            if t >= 1:
                continue
            x = x1 + (x2 - x1) * t
            y = y1 + (y2 - y1) * t
            ax.add_patch(
                patches.Circle(
                    (x, y),
                    radius=width_mm / 140,
                    facecolor="white",
                    edgecolor=BELT_COLOR_EDGE,
                    linewidth=1,
                    zorder=7,
                )
            )

def save_design_plot_png(rooms: List[Room], flows, find_machine_func, routing_func, out_png="design_plot.png", use_processed_images=True):
    """
    Save design plot as PNG:
      - Images are auto-cropped to remove whitespace.
      - Images are scaled to 15% of layout width (unless constrained by neighbors).
      - Respects 'no overlap' constraint by checking nearest neighbors.
      - use_processed_images: If True (default), uses _nobg.png versions if available.
                              If False, uses raw cached images (for AI inference input).
    """
    
    # 1. Collect all machines
    all_machines = []
    for r in rooms:
        for p in r.machines:
            all_machines.append(p)
            
    # Calculate layout bounds (physical)
    phys_rects = [rect_bounds(m.x, m.y, m.w, m.h) for m in all_machines]
    p_minx, p_miny, p_maxx, p_maxy = compute_bounds_rects(phys_rects)
    layout_width = max(1.0, p_maxx - p_minx)
    
    # HEURISTIC: Target Size = 15% of total layout width
    # This is roughly 2x bigger than previous attempts
    IDEAL_IMG_SIZE = max(3000.0, layout_width * 0.15) 
    
    # 2. Determine Image Sizes based on nearest neighbor
    draw_items = []
    visual_rects = [] 

    for i, p in enumerate(all_machines):
        # Calculate distance to nearest neighbor to prevent overlap
        min_dist = float('inf')
        cx = p.x + p.w / 2.0
        cy = p.y + p.h / 2.0
        
        for j, other in enumerate(all_machines):
            if i == j: continue
            ocx = other.x + other.w / 2.0
            ocy = other.y + other.h / 2.0
            dist = math.hypot(cx - ocx, cy - ocy)
            if dist < min_dist:
                min_dist = dist
        
        # Determine allowed size
        # We allow the image to fill 95% of the gap to the nearest neighbor.
        if min_dist == float('inf'):
            allowed_size = IDEAL_IMG_SIZE * 1.5 # Isolated machines can be even bigger
        else:
            allowed_size = min_dist * 0.95
            
        # Target size is the lesser of the Ideal Size (visual consistency) and Allowed Size (geometry constraint)
        # We also enforce a hard floor (e.g. 1000mm) so machines don't vanish even if they are touching.
        target_size = max(1000.0, min(IDEAL_IMG_SIZE, allowed_size))

        if p.img_path and os.path.exists(p.img_path):
            try:
                # Determine which image to use
                img_to_load = p.img_path
                if use_processed_images:
                    # Check if a _nobg.png version exists
                    base, ext = os.path.splitext(p.img_path)
                    nobg_path = f"{base}_nobg.png"
                    if os.path.exists(nobg_path):
                        img_to_load = nobg_path
                
                im = Image.open(img_to_load).convert("RGBA")
                im = ImageOps.exif_transpose(im)
                
                # CRITICAL: Trim whitespace so the "machine" fills the box
                im = trim_whitespace(im)
                
                # Calculate aspect ratio
                iw_px, ih_px = im.size
                aspect = iw_px / ih_px
                
                # Fit into bounding box defined by target_size
                if aspect > 1:
                    draw_w = target_size
                    draw_h = target_size / aspect
                else:
                    draw_h = target_size
                    draw_w = target_size * aspect
                
                # Center image on the machine's center
                draw_x = cx - (draw_w / 2.0)
                draw_y = cy - (draw_h / 2.0)
                
                draw_items.append({
                    "im": im,
                    "extent": [draw_x, draw_x + draw_w, draw_y, draw_y + draw_h],
                    "z": 10
                })
                # Add to bounds so plot zooms out to fit large images
                visual_rects.append((draw_x, draw_y, draw_x + draw_w, draw_y + draw_h))
            except Exception as e:
                print(f"Error loading image {p.img_path}: {e}")
                visual_rects.append((p.x, p.y, p.x + p.w, p.y + p.h))
        else:
            visual_rects.append((p.x, p.y, p.x + p.w, p.y + p.h))

    # --- Setup Plot Bounds ---
    if not visual_rects:
        visual_rects = [rect_bounds(m.x, m.y, m.w, m.h) for m in all_machines]
        
    minx, miny, maxx, maxy = compute_bounds_rects(visual_rects)

    # Add 5% padding
    pad_w = (maxx - minx) * 0.05
    pad_h = (maxy - miny) * 0.05
    minx -= pad_w
    maxx += pad_w
    miny -= pad_h
    maxy += pad_h

    # Enforce 1:1 square aspect ratio
    current_w = maxx - minx
    current_h = maxy - miny
    target_aspect = 1.0  # 1:1 square aspect ratio

    if current_w / current_h > target_aspect:
        target_h = current_w / target_aspect
        diff = target_h - current_h
        miny -= diff / 2
        maxy += diff / 2
    else:
        target_w = current_h * target_aspect
        diff = target_w - current_w
        minx -= diff / 2
        maxx += diff / 2

    # --- Plotting ---
    fig = plt.figure(figsize=(16, 9), dpi=100)
    ax = fig.add_subplot(111)
    ax.set_facecolor("#00FF00")
    fig.patch.set_facecolor("#00FF00")

    # Draw Conveyors
    for src, dst in flows:
        a = find_machine_func(rooms, src)
        b = find_machine_func(rooms, dst)
        if not a or not b:
            continue
        pts = routing_func(a, b)

    for item in draw_items:
        ax.imshow(item["im"], extent=item["extent"], zorder=item["z"])

    ax.set_xlim(minx, maxx)
    ax.set_ylim(miny, maxy)
    ax.set_aspect("equal", adjustable="box")
    ax.axis("off")
    plt.tight_layout(pad=0)
    fig.savefig(out_png, bbox_inches="tight", pad_inches=0)
    plt.close(fig)
    print(f"[+] Saved: {out_png}")

"""
Edge-based routing with A* pathfinding (no crossings by construction).
"""

import heapq
from typing import Dict, List, Optional, Tuple, Set

from models import Placed
from grid import OccupancyGrid
from config import (
    EDGE_CAPACITY_DEFAULT, EDGE_CAPACITY_TRUNK, BEND_PENALTY,
    CONGESTION_PENALTY, MAX_ASTAR_ITERS, KEEP_OUT, PORT_OFFSET
)

# Edge keys:
# H edge between (x,y)-(x+1,y): ("H", x, y)
# V edge between (x,y)-(x,y+1): ("V", x, y)

EdgeKey = Tuple[str, int, int]
Node = Tuple[int, int]
Path = List[Node]


class EdgeReservation:
    def __init__(self):
        self.use: Dict[EdgeKey, int] = {}
        self.cap: Dict[EdgeKey, int] = {}

    def capacity(self, e: EdgeKey) -> int:
        return self.cap.get(e, EDGE_CAPACITY_DEFAULT)

    def set_capacity_rect_trunk(self, x1: int, y1: int, x2: int, y2: int) -> None:
        """Mark a trunk corridor rectangle edges with higher capacity to allow merges."""
        for y in range(y1, y2+1):
            for x in range(x1, x2):
                self.cap[("H", x, y)] = EDGE_CAPACITY_TRUNK
        for x in range(x1, x2+1):
            for y in range(y1, y2):
                self.cap[("V", x, y)] = EDGE_CAPACITY_TRUNK

    def can_use(self, e: EdgeKey) -> bool:
        return self.use.get(e, 0) < self.capacity(e)

    def add_path(self, path: Path) -> List[EdgeKey]:
        edges = nodes_to_edges(path)
        for e in edges:
            self.use[e] = self.use.get(e, 0) + 1
        return edges

    def remove_edges(self, edges: List[EdgeKey]) -> None:
        for e in edges:
            if e in self.use:
                self.use[e] -= 1
                if self.use[e] <= 0:
                    del self.use[e]


def nodes_to_edges(path: Path) -> List[EdgeKey]:
    out: List[EdgeKey] = []
    for (x1, y1), (x2, y2) in zip(path, path[1:]):
        if x2 == x1 + 1 and y2 == y1:
            out.append(("H", x1, y1))
        elif x2 == x1 - 1 and y2 == y1:
            out.append(("H", x2, y1))
        elif y2 == y1 + 1 and x2 == x1:
            out.append(("V", x1, y1))
        elif y2 == y1 - 1 and x2 == x1:
            out.append(("V", x1, y2))
        else:
            raise ValueError("Non-grid step in path")
    return out


def edge_between(a: Node, b: Node) -> EdgeKey:
    x1, y1 = a
    x2, y2 = b
    if x2 == x1 + 1 and y2 == y1:
        return ("H", x1, y1)
    if x2 == x1 - 1 and y2 == y1:
        return ("H", x2, y1)
    if y2 == y1 + 1 and x2 == x1:
        return ("V", x1, y1)
    if y2 == y1 - 1 and x2 == x1:
        return ("V", x1, y2)
    raise ValueError("Not adjacent")


def manhattan(a: Node, b: Node) -> int:
    return abs(a[0]-b[0]) + abs(a[1]-b[1])


def astar_route(
    grid: OccupancyGrid,
    res: EdgeReservation,
    start: Node,
    goal: Node,
) -> Optional[Path]:
    """A* on node graph with edge capacity constraints."""
    n_cells = grid.n
    if not (0 <= start[0] < n_cells and 0 <= start[1] < n_cells):
        return None
    if not (0 <= goal[0] < n_cells and 0 <= goal[1] < n_cells):
        return None
    if grid.block[start[1]][start[0]] != 0:
        return None
    if grid.block[goal[1]][goal[0]] != 0:
        return None

    openh = []
    gscore: Dict[Node, float] = {start: 0.0}
    parent: Dict[Node, Node] = {}
    dir_from: Dict[Node, Optional[Tuple[int,int]]] = {start: None}

    heapq.heappush(openh, (manhattan(start, goal), 0.0, start))
    iters = 0

    while openh:
        iters += 1
        if iters > MAX_ASTAR_ITERS:
            return None
        _, gc, cur = heapq.heappop(openh)
        if cur == goal:
            # reconstruct
            path = [cur]
            while cur in parent:
                cur = parent[cur]
                path.append(cur)
            path.reverse()
            return path

        for nb in [(cur[0]+1, cur[1]), (cur[0]-1, cur[1]), (cur[0], cur[1]+1), (cur[0], cur[1]-1)]:
            x, y = nb
            if x < 0 or y < 0 or x >= n_cells or y >= n_cells:
                continue
            if grid.block[y][x] != 0:
                continue

            # edge capacity check
            e = edge_between(cur, nb)
            if not res.can_use(e):
                continue

            # cost: step + bend + congestion
            step = 1.0
            dcur = (nb[0]-cur[0], nb[1]-cur[1])
            dprev = dir_from.get(cur)
            bend = 0.0 if (dprev is None or dprev == dcur) else BEND_PENALTY
            cong = CONGESTION_PENALTY * res.use.get(e, 0)

            ng = gc + step + bend + cong
            if ng < gscore.get(nb, 1e18):
                gscore[nb] = ng
                parent[nb] = cur
                dir_from[nb] = dcur
                f = ng + manhattan(nb, goal)
                heapq.heappush(openh, (f, ng, nb))

    return None


def pick_port(src: Placed, dst: Placed) -> Node:
    """Choose a start cell near src facing dst."""
    sx, sy = src.center()
    dx, dy = dst.center()
    # direction preference
    if abs(dx - sx) > abs(dy - sy):
        # go left/right
        if dx > sx:
            px = min(src.x + src.w + PORT_OFFSET, src.x + src.w - 1)
            py = int(round(src.y + src.h / 2))
            return (min(px, src.x + src.w - 1), min(max(py, src.y), src.y + src.h - 1))
        else:
            px = max(src.x - 1 - PORT_OFFSET, src.x)
            py = int(round(src.y + src.h / 2))
            return (max(px, src.x), min(max(py, src.y), src.y + src.h - 1))
    else:
        # go up/down
        if dy > sy:
            px = int(round(src.x + src.w / 2))
            py = min(src.y + src.h + PORT_OFFSET, src.y + src.h - 1)
            return (min(max(px, src.x), src.x + src.w - 1), min(py, src.y + src.h - 1))
        else:
            px = int(round(src.x + src.w / 2))
            py = max(src.y - 1 - PORT_OFFSET, src.y)
            return (min(max(px, src.x), src.x + src.w - 1), max(py, src.y))


def port_outside_cell(p: Placed, towards: Node, grid: OccupancyGrid) -> Node:
    """Move from inside machine to nearest free cell adjacent to its boundary."""
    cx, cy = p.center()
    tx, ty = towards
    best = None
    bestd = 1e18
    for y in range(max(0, p.y - (KEEP_OUT+2)), min(grid.n, p.y + p.h + (KEEP_OUT+2))):
        for x in range(max(0, p.x - (KEEP_OUT+2)), min(grid.n, p.x + p.w + (KEEP_OUT+2))):
            if grid.block[y][x] == 0:
                d = abs(x - tx) + abs(y - ty) + 0.2*(abs(x - cx) + abs(y - cy))
                if d < bestd:
                    bestd = d
                    best = (x, y)
    return best if best else (int(cx), int(cy))


def route_all_flows(
    grid: OccupancyGrid,
    placed: List[Placed],
    flows: List[Tuple[str,str]],
) -> Tuple[EdgeReservation, Dict[Tuple[str,str], Path], Dict[Tuple[str,str], List[EdgeKey]]]:
    """Route all flows with A*."""
    by_id = {p.uid: p for p in placed}
    res = EdgeReservation()

    # optional: create a central trunk corridor to encourage merges
    n = grid.n
    mid = n // 2
    res.set_capacity_rect_trunk(max(0, mid-2), 0, min(n-1, mid+2), n-1)

    paths: Dict[Tuple[str,str], Path] = {}
    edges_for: Dict[Tuple[str,str], List[EdgeKey]] = {}

    # route in hardest-first: longer estimated dist first
    scored = []
    for a, b in flows:
        if a not in by_id or b not in by_id:
            continue
        da = manhattan((int(by_id[a].center()[0]), int(by_id[a].center()[1])),
                       (int(by_id[b].center()[0]), int(by_id[b].center()[1])))
        scored.append((da, a, b))
    scored.sort(reverse=True)

    for _, a, b in scored:
        pa, pb = by_id[a], by_id[b]
        approx = (int(pb.center()[0]), int(pb.center()[1]))
        sa = port_outside_cell(pa, approx, grid)
        sb = port_outside_cell(pb, (int(pa.center()[0]), int(pa.center()[1])), grid)

        path = astar_route(grid, res, sa, sb)
        if path is None:
            return res, paths, edges_for  # partial
        edges = res.add_path(path)
        paths[(a,b)] = path
        edges_for[(a,b)] = edges

    return res, paths, edges_for


def ripup_reroute(
    grid: OccupancyGrid,
    placed: List[Placed],
    flows: List[Tuple[str,str]],
) -> Tuple[EdgeReservation, Dict[Tuple[str,str], Path]]:
    """Iterative ripup and reroute to handle congestion."""
    from config import RIPUP_PASSES
    
    by_id = {p.uid: p for p in placed}

    # initial attempt
    res, paths, edges_for = route_all_flows(grid, placed, flows)
    if len(paths) == len([f for f in flows if f[0] in by_id and f[1] in by_id]):
        return res, paths

    # iterative ripup
    for _pass in range(RIPUP_PASSES):
        target_flows = [(a,b) for (a,b) in flows if a in by_id and b in by_id]
        if len(paths) == len(target_flows):
            return res, paths

        unrouted = [f for f in target_flows if f not in paths]
        if not unrouted:
            return res, paths

        # rip up longest routes
        routed = list(paths.keys())
        routed.sort(key=lambda k: len(paths[k]), reverse=True)
        k_remove = min(6, max(2, len(routed)//6))
        to_remove = routed[:k_remove]

        for key in to_remove:
            res.remove_edges(edges_for[key])
            del edges_for[key]
            del paths[key]

        # fresh reroute
        fresh = EdgeReservation()
        n = grid.n
        mid = n // 2
        fresh.set_capacity_rect_trunk(max(0, mid-2), 0, min(n-1, mid+2), n-1)

        res = fresh
        paths = {}
        edges_for = {}

        scored = []
        for x, y in target_flows:
            da = manhattan((int(by_id[x].center()[0]), int(by_id[x].center()[1])),
                           (int(by_id[y].center()[0]), int(by_id[y].center()[1])))
            scored.append((da, x, y))
        scored.sort(reverse=True)

        ok_all = True
        for _, x, y in scored:
            px, py = by_id[x], by_id[y]
            sa = port_outside_cell(px, (int(py.center()[0]), int(py.center()[1])), grid)
            sb = port_outside_cell(py, (int(px.center()[0]), int(px.center()[1])), grid)
            path = astar_route(grid, res, sa, sb)
            if path is None:
                ok_all = False
                break
            edges = res.add_path(path)
            paths[(x,y)] = path
            edges_for[(x,y)] = edges

        if ok_all:
            return res, paths

    return res, paths

def ask_llm(prompt: str, system_prompt: str = None, temperature: float = 0.7, max_tokens: int = 8000) -> str:
    """
    Send a prompt to Groq's Gemini Flash model and get a response.
    
    Args:
        prompt: The user prompt to send to the LLM
        system_prompt: Optional system prompt to guide the LLM's behavior
        temperature: Sampling temperature (0.0-2.0)
        max_tokens: Maximum tokens in the response
        
    Returns:
        The LLM's response as a string
    """
    # Get API key from environment
    API_KEY = os.getenv("GROQ_API_KEY")
    if not API_KEY:
        raise ValueError("GROQ_API_KEY not found in environment variables")
    
    # Initialize Groq client
    client = Groq(api_key=API_KEY)
    
    # Prepare messages
    messages = []
    if system_prompt:
        messages.append({
            "role": "system",
            "content": system_prompt
        })
    messages.append({
        "role": "user",
        "content": prompt
    })
    
    # Make API call
    try:
        chat_completion = client.chat.completions.create(
            messages=messages,
            model="llama-3.1-8b-instant",  # Groq's Llama 3.1 8B Instant model
            temperature=temperature,
            max_tokens=max_tokens,
        )
        
        return chat_completion.choices[0].message.content
    
    except Exception as e:
        print(f"Error calling Groq API: {e}")
        raise


def ask_llm_json(prompt: str, system_prompt: str = None, temperature: float = 0.7) -> dict:
    """
    Send a prompt to the LLM and parse the response as JSON.
    
    Args:
        prompt: The user prompt to send to the LLM
        system_prompt: Optional system prompt to guide the LLM's behavior
        temperature: Sampling temperature (0.0-2.0)
        
    Returns:
        The LLM's response parsed as a dictionary
    """
    response = ask_llm(prompt, system_prompt, temperature)
    
    # Try to extract JSON from markdown code blocks if present
    if "```json" in response:
        start = response.find("```json") + 7
        end = response.find("```", start)
        response = response[start:end].strip()
    elif "```" in response:
        start = response.find("```") + 3
        end = response.find("```", start)
        response = response[start:end].strip()
    
    return json.loads(response)

image_client = genai.Client(api_key=os.getenv("IMAGE_API_KEY"))
video_client = genai.Client(api_key=os.getenv("VIDEO_API_KEY"))

def generate_message(
    prompt: str,
    system_instruction: Optional[str] = None,
    tools: Optional[List] = None,
    images: Optional[List[Union[bytes, str]]] = None,
    documents: Optional[List[Union[bytes, str]]] = None,
    videos: Optional[List[Union[bytes, str]]] = None,
    audio: Optional[List[Union[bytes, str]]] = None,
    temperature: float = 1.0,
    top_p: float = 0.95,
) -> str:
    """
    Generates text using Gemini 2.5 Flash with multimodal support.
    
    Supports:
    - Images: png, jpeg, webp, heic, heif (max 3,000 per prompt)
    - Documents: pdf, text/plain (max 3,000 per prompt)
    - Videos: mp4, webm, quicktime, etc. (max 10 per prompt)
    - Audio: mp3, wav, flac, etc. (max 1 per prompt)
    
    Args:
        prompt: The text prompt to send
        system_instruction: Optional system instruction
        tools: Optional list of tools/functions
        images: Optional list of image inputs (MediaInput, bytes, base64 str, or dict with 'data' and 'mime_type')
        documents: Optional list of document inputs
        videos: Optional list of video inputs  
        audio: Optional list of audio inputs
        config: Optional MessageConfig for generation parameters
        
    Returns:
        Generated text response
    """
    model_id = os.getenv("GEMINI_MODEL", "gemini-2.5-flash")
    
    # Build contents with media
    contents = []
    
    def add_media(items, mime):
        if items:
            for item in items:
                if isinstance(item, str) and item.startswith("gs://"):
                    contents.append(types.Part.from_uri(file_uri=item, mime_type=mime))
                else:
                    data = base64.b64decode(item) if isinstance(item, str) else item
                    contents.append(types.Part.from_bytes(data=data, mime_type=mime))
    
    add_media(images, "image/jpeg")
    add_media(documents, "application/pdf")
    add_media(videos, "video/mp4")
    add_media(audio, "audio/mp3")
    contents.append(prompt)
    
    gen_config = types.GenerateContentConfig(
        system_instruction=system_instruction,
        tools=tools,
        temperature=temperature,
        top_p=top_p,
        top_k=64,
    )

    response = gemini_client.models.generate_content(
        model=model_id,
        contents=contents,
        config=gen_config,
    )
    
    return response.text

def generate_image(
    prompt: str,
    images: Optional[List[Union[bytes, str]]] = None,
    candidate_count: int = 1,
    system_instruction: Optional[str] = None,
    temperature: float = 1.0,
    top_p: float = 0.95,
) -> List[bytes]:
    """
    Generates images using gemini-3-pro-image-preview.
    
    Args:
        prompt: Text description of the image to generate
        images: Optional input images (bytes or base64 str) for editing, max 14
        system_instruction: Optional system instruction
        temperature: 0.0-2.0 (default 1.0)
        top_p: 0.0-1.0 (default 0.95)
        
    Returns:
        List of generated image bytes
    """
    import time
    from google.genai.errors import ServerError

    model_id = os.getenv("IMAGE_MODEL", "gemini-3-pro-image-preview")
    
    contents = []
    if images:
        for img in images[:14]:
            img_bytes = base64.b64decode(img) if isinstance(img, str) else img
            contents.append(types.Part.from_bytes(data=img_bytes, mime_type="image/jpeg"))
    contents.append(prompt)

    max_retries = 3
    for attempt in range(max_retries):
        try:
            response = image_client.models.generate_content(
                model=model_id,
                contents=contents,
                config=types.GenerateContentConfig(
                    system_instruction=system_instruction,
                    response_modalities=["IMAGE", "TEXT"],
                    temperature=temperature,
                    top_p=top_p,
                    top_k=64,
                    candidate_count=candidate_count,
                )
            )
            
            result = []
            for part in response.candidates[0].content.parts:
                if part.inline_data and part.inline_data.mime_type.startswith("image/"):
                    result.append(part.inline_data.data)
            return result
            
        except ServerError as e:
            if attempt < max_retries - 1:
                print(f"⚠️  Image generation server error (500), retrying in 5s... (Attempt {attempt+1}/{max_retries})")
                time.sleep(5)
                continue
            else:
                print(f"❌ Image generation failed after {max_retries} attempts.")
                raise e
        except Exception as e:
            print(f"❌ Image generation failed with error: {e}")
            raise e
    return []

def remove_background(image_bytes: bytes) -> bytes:
    """
    Smart background removal:
    - If chroma-green is detected -> pure chroma key
    - Otherwise -> fallback to rembg
    - ALWAYS convert remaining green pixels to grey
    """

    import io
    import numpy as np
    from PIL import Image
    from rembg import remove

    img = Image.open(io.BytesIO(image_bytes)).convert("RGBA")
    arr = np.array(img)

    r, g, b, a = arr[..., 0], arr[..., 1], arr[..., 2], arr[..., 3]

    # --- Detect green screen coverage ---
    green_mask = (g > 200) & (r < 80) & (b < 80)
    green_ratio = green_mask.mean()

    # --- If image is mostly green → CHROMA KEY ---
    if green_ratio > 0.15:
        arr[green_mask, 3] = 0  # transparent alpha
        out = Image.fromarray(arr, "RGBA")
    else:
        out = remove(img).convert("RGBA")

    # --- FORCE remove any remaining green pixels (turn to grey) ---
    arr = np.array(out)
    r, g, b, a = arr[..., 0], arr[..., 1], arr[..., 2], arr[..., 3]
    green_left = (g > 150) & (r < 120) & (b < 120) & (a > 0)
    arr[green_left] = [200, 200, 200, a[green_left]]
    out = Image.fromarray(arr, "RGBA")

    buf = io.BytesIO()
    out.save(buf, format="PNG")
    return buf.getvalue()
